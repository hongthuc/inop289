    // Điều khiển Slider 2 Banner tự trượt bằng kích thước
    function showBannerSlide(index) {
      const slide0 = document.getElementById('banner-slide-0');
      const slide1 = document.getElementById('banner-slide-1');
      const dot0 = document.getElementById('banner-dot-0');
      const dot1 = document.getElementById('banner-dot-1');
      if (!slide0 || !slide1) return;

      currentBannerIndex = index;
      if (index === 0) {
        slide0.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
        slide0.classList.add('opacity-100', 'z-10');
        slide1.classList.remove('opacity-100', 'z-10');
        slide1.classList.add('opacity-0', 'z-0', 'pointer-events-none');
        dot0.className = 'w-6 h-2 rounded-full bg-red-600 transition-all';
        dot1.className = 'w-2.5 h-2 rounded-full bg-white/60 hover:bg-white transition-all';
      } else {
        slide1.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
        slide1.classList.add('opacity-100', 'z-10');
        slide0.classList.remove('opacity-100', 'z-10');
        slide0.classList.add('opacity-0', 'z-0', 'pointer-events-none');
        dot1.className = 'w-6 h-2 rounded-full bg-red-600 transition-all';
        dot0.className = 'w-2.5 h-2 rounded-full bg-white/60 hover:bg-white transition-all';
      }
    }

    function changeBannerSlide(step) {
      const next = (currentBannerIndex + step + 2) % 2;
      showBannerSlide(next);
      restartBannerAutoSlide();
    }

    function goToBannerSlide(index) {
      showBannerSlide(index);
      restartBannerAutoSlide();
    }

    function restartBannerAutoSlide() {
      if (bannerTimer) clearInterval(bannerTimer);
      bannerTimer = setInterval(() => {
        const next = (currentBannerIndex + 1) % 2;
        showBannerSlide(next);
      }, 4500);
    }

    // Giỏ hàng & Modal
    function addToCart(productId) {
      const product = DRIVE_PRODUCTS.find(p => p.id === productId);
      if (!product) return;
      const existing = cartItems.find(item => item.id === productId);
      if (existing) {
        existing.qty += 1;
      } else {
        cartItems.push({ ...product, qty: 1 });
      }
      updateCartBadge();
      openCartModal();
    }

      const total = cartItems.reduce((sum, item) => sum + item.qty, 0);
      document.getElementById('cart-count-badge').textContent = total;
